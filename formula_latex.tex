\section*{爆炸冲击波超压峰值经验公式}

\subsection*{变量定义}
\begin{align*}
  V_n &= \frac{V}{1000} \quad (\text{归一化速度, m/s}) \\
  T_h &= T_0 - 6.5H \quad (\text{海拔修正温度, K}) \\
  S_p &= \left(\frac{T_h}{T_0}\right)^{5.25588} \quad (\text{萨克斯缩放因子}) \\
  T_0 &= 288.15\,\mathrm{K} \quad (\text{海平面标准温度})
\end{align*}

\subsection*{通用公式结构}
\begin{equation}
  P_{\mathrm{dyn}}(Z, S_p, \theta, V) = T_1 + T_2 + T_3 + B \cdot \mathbf{1}_{Z=4}
\end{equation}
\begin{align*}
  T_1 &= \frac{S_p^{2/3}}{Z} \sum_{k=0}^{6} (a_{1,k} + b_{1,k} V_n) \cos(k\theta) \\
  T_2 &= \frac{S_p^{1/3}}{Z^2} \sum_{k=0}^{6} (a_{2,k} + b_{2,k} V_n) \cos(k\theta) \\
  T_3 &= \frac{1}{Z^3} \sum_{k=0}^{6} (a_{3,k} + b_{3,k} V_n) \cos(k\theta)
\end{align*}

\textbf{使用规则:}
\begin{itemize}
  \item $Z \leq 2.5$: 使用段S1系数 (中近场, Z=1,2)
  \item $2.5 < Z \leq 3.5$: 使用段S2系数 (中场, Z=3)
  \item $Z > 3.5$: 使用段S3系数 (远场边界, Z=4, 含前向增强项B)
\end{itemize}

\textbf{前向增强项 (仅Z=4):}
\begin{equation}
  B = \frac{S_p^{2/3}}{Z}\left[V_n\left(0.5172-0.0703 f(\theta)-0.1512 f^2(\theta)\right) + V_n^2\left(-0.0600-0.2667 f(\theta)\right)\right]
  + \frac{1}{Z^3}\left[V_n\left(-1.0814-0.7139 f(\theta)-0.5373 f^2(\theta)\right) + V_n^2\left(1.2077+3.7963 f(\theta)\right)\right]
\end{equation}
其中 $f(\theta) = \frac{1+\cos\theta}{2}$ 为前向定向函数。
